CMD ["sh", "-c", "
echo '=== Running linters ==='
black --check . && echo '✅ Black: OK' || echo '⚠ Black: Needs formatting'
flake8 . && echo '✅ Flake8: OK' || echo '⚠ Flake8: Issues found'

echo '=== Running security checks ==='
bandit -r . -l && echo '✅ Bandit: Security check completed'
safety check && echo '✅ Safety: Dependencies check completed'

echo '=== Running tests ==='
python -m pytest test_main.py -v || echo '⚠ Pytest failed'

echo '=== Starting FastAPI application ==='
uvicorn main:app --host 0.0.0.0 --port 8000
"]
